**File:** `client/src/main.jsx` or `client/src/index.js`

import { LocalizationProvider } from '@mui/x-date-pickers/LocalizationProvider';

### Step 5: Use Modals in Your Pages

```javascript
import { useState } from 'react';
import { Button } from '@mui/material';
import TransactionModal from './components/TransactionModal';
import CostItemModal from './components/CostItemModal';
import InventoryUsageModal from './components/InventoryUsageModal';
import { useTranslation } from 'react-i18next';

function ProjectPage({ projectId, currentUser }) {
  const { t } = useTranslation();
  const [transactionModalOpen, setTransactionModalOpen] = useState(false);
  const [costItemModalOpen, setCostItemModalOpen] = useState(false);
  const [inventoryModalOpen, setInventoryModalOpen] = useState(false);

  const handleTransactionSubmit = async (formData) => {
    try {
      const response = await fetch('/api/transactions', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${localStorage.getItem('token')}`
        },
        body: JSON.stringify(formData)
      });
      
      const data = await response.json();
      
      if (data.success) {
        console.log('Transaction created:', data.data);
        console.log('Profit info:', data.profitInfo);
        // Refresh data
      }
    } catch (error) {
      console.error('Error:', error);
    }
  };

  const handleCostItemSuccess = (newItem) => {
    console.log('Cost item created:', newItem);
    // Refresh cost items list
  };

  const handleInventorySuccess = (usageData, profitInfo) => {
    console.log('Usage recorded:', usageData);
    console.log('Profit recognized:', profitInfo.profit);
    // Refresh inventory
  };

  return (
    <div>
      <Button onClick={() => setTransactionModalOpen(true)}>
        {t('transaction.title')}
      </Button>
      
      <Button onClick={() => setCostItemModalOpen(true)}>
        {t('costItem.title')}
      </Button>
      
      <Button onClick={() => setInventoryModalOpen(true)}>
        {t('inventory.title')}
      </Button>

      <TransactionModal
        open={transactionModalOpen}
        onClose={() => setTransactionModalOpen(false)}
        onSubmit={handleTransactionSubmit}
        projectId={projectId}
        currentUser={currentUser}
      />

      <CostItemModal
        open={costItemModalOpen}
        onClose={() => setCostItemModalOpen(false)}
        projectId={projectId}
        onSuccess={handleCostItemSuccess}
      />

      <InventoryUsageModal
        open={inventoryModalOpen}
        onClose={() => setInventoryModalOpen(false)}
        projectId={projectId}
        onSuccess={handleInventorySuccess}
      />
    </div>
  );
}

export default ProjectPage;
```

## 🔧 Customization

### Adding New Languages

1. Create new locale file: `client/src/locales/fr.js`
2. Add to i18n config:
```javascript
import fr from './locales/fr';

i18n.init({
  resources: {
    en: { translation: en },
    ar: { translation: ar },
    fr: { translation: fr }
  }
});
```
3. Add to LanguageSwitcher:
```javascript
const languages = [
  { code: 'en', name: 'English', nativeName: 'English' },
  { code: 'ar', name: 'Arabic', nativeName: 'العربية' },
  { code: 'fr', name: 'French', nativeName: 'Français' }
];
```

### Customizing Validation

Edit validation functions in each modal:
```javascript
const validate = () => {
  const newErrors = {};
  
  // Add your custom validation rules
  if (customCondition) {
    newErrors.field = 'Custom error message';
  }
  
  return Object.keys(newErrors).length === 0;
};
```

## 🆘 Troubleshooting

### Issue: Translations not showing
**Solution:** Ensure i18n is imported in main.jsx/index.js before App

### Issue: RTL not working
**Solution:** Check that `dir` attribute is set on Dialog component

### Issue: Date picker in wrong language
**Solution:** Import locale from date-fns and pass to LocalizationProvider

### Issue: Categories not loading
**Solution:** Verify `/api/categories` endpoint is accessible and returns correct format
